{{- /* A relative path is a resource of the page, else a file of site/assets/: a screenshot taken at twice its size, shown at half its pixels. A URL or an absolute path is used as written. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := .Destination -}} {{- $img := false -}} {{- if not (or $u.IsAbs (hasPrefix $u.Path "/")) -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $img = . -}} {{- $src = .RelPermalink -}} {{- else -}} {{- errorf "%s: no image %q in the page or site/assets/" .Page.Path .Destination -}} {{- end -}} {{- end -}}
